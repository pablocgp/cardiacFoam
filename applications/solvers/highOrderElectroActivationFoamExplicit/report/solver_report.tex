\title{Solver Specification:\\
       \texorpdfstring{\texttt{highOrderElectroActivationFoamExplicit}}{highOrderElectroActivationFoamExplicit}\\
       \large A high-order, explicit forward-Euler
       monodomain solver in OpenFOAM with CFL-bounded time step}
\author{Pablo}
\date{\today}

\begin{document}
\maketitle

\begin{abstract}
This report specifies the algorithms implemented in the OpenFOAM solver:

\begin{center}
\code{highOrderElectroActivationFoamExplicit}.
\end{center}

The solver advances the monodomain equations of cardiac
electrophysiology coupled to the ten Tusscher--Noble--Noble--Panfilov
(TNNP, 2004) ionic cell model. The diffusion operator is discretised
with the cell-centred finite-volume framework of OpenFOAM, optionally
upgraded to a Local Regression Estimators (LRE) high-order
reconstruction; the ionic source is evaluated either at cell centres
or at cell quadrature points; time integration is the explicit
forward-Euler scheme, with a stable time step bounded by the
parabolic CFL condition computed at run time from the local
diffusivity and the characteristic cell size. A staggered
operator-split structures the per-step work: the algebraic ionic
current at $t^{n}$ is updated first, then the diffusion Laplacian is
assembled, the voltage is advanced by an Euler step, and finally the
TNNP cell ODEs are integrated from $t^{n}$ to $t^{n+1}$ using the
just-updated voltage. The per-cell ionic ODE uses OpenFOAM's
adaptive RKF45 integrator with configurable tolerances; the TNNP
cell model is wrapped in a phase-aware \code{protectState} guard that
enforces physical bounds before every algebraic evaluation;
activation times are recorded by linear interpolation across the
first threshold crossing and benchmark points are sampled by
inverse-distance weighting. The report also discusses how the
CFL-bounded explicit scheme behaves on the 2D and 3D Niederer-style
benchmark runs: in
particular, how the reported \emph{requested} $\Delta t$ relates to
the \emph{effective} $\Delta t$ the solver actually uses after the
CFL clamp. All algorithmic parameters are exposed through the
\code{constant/spatialIntegrationProperties} and
\code{constant/timeIntegrationProperties} dictionaries, and the
companion Python driver \code{run\_convergence.py} generates these
dictionaries from a single configuration file. Section~\ref{sec:results}
collects measured P8 activation times, wall-clock cost and peak
memory from a 2D Niederer-style sweep and a sparser 3D Niederer-style
sweep.
\end{abstract}

\section{Governing equations}
% ================================================================== %

\subsection{Monodomain PDE}

The bulk voltage is propagated by the standard monodomain
reaction--diffusion equation~\cite{ttp2004,niederer2012}:
\begin{equation}
  \chi \, C_{m} \, \frac{\partial \Vm}{\partial t}
   \;=\;
   \divg\!\bigl( \Dten \grad \Vm \bigr)
   - \Iion(\Vm, \mathbf{s})
   + \Iext
   \quad \text{in } \Omega \times (0, T],
   \label{eq:monodomain}
\end{equation}
subject to insulated boundaries
$\mathbf{n} \cdot ( \Dten \grad \Vm ) = 0$ on
$\partial \Omega \times (0, T]$, and initial condition
$\Vm(\mathbf{x}, 0) = -\SI{84}{\milli\volt}$. The conductivity tensor
$\Dten$ is symmetric positive-definite and either uniform or read as a
\code{volTensorField} from disk.

Dividing~\eqref{eq:monodomain} by $\chi C_{m}$ gives the form actually
used inside the solver:
\begin{equation}
  \frac{\partial \Vm}{\partial t}
   \;=\;
   \mathcal{L} \Vm
   \;+\;
   s(\Vm, \mathbf{s}),
   \qquad
   \mathcal{L} \;=\; \frac{1}{\chi C_{m}}\,\divg(\Dten \grad \cdot),
   \quad
   s \;=\; -\frac{\Iion}{\chi C_{m}} + \frac{\Iext}{\chi C_{m}}.
   \label{eq:semi-discrete}
\end{equation}

\subsection{Ionic ODE}

At every spatial sampling point the TNNP state vector
$\mathbf{s} \in \mathbb{R}^{17}$ evolves according to
\begin{equation}
  \frac{\mathrm{d}\mathbf{s}}{\mathrm{d}t}
   \;=\;
   \mathbf{f}_{\text{TNNP}}\!\bigl(\Vm, \mathbf{s}; \texttt{tissue}\bigr),
   \qquad \mathbf{s}(0) = \mathbf{s}_{0},
   \label{eq:tnnp}
\end{equation}
where \code{tissue} selects \code{epicardialCells}, \code{mCells} or
\code{endocardialCells}. The 17 tracked states are
\[
  \mathbf{s} = (V,\ K_{i},\ Na_{i},\ Ca_{i},\ Xr_{1},\ Xr_{2},\ Xs,\ m,\ h,\ j,\ d,\ f,\ fCa,\ s,\ r,\ g,\ Ca_{SR}).
\]
The total ionic current returned to the PDE is \code{Iion\_cm} of
TNNP~\cite{ttp2004}; the full rate-equation set is reproduced in
Appendix~\ref{app:tnnp} for reference.

\paragraph{Units and sign convention.} The field \code{Iion\_cm}
returned by the TNNP wrapper is the per-capacitance ionic current,
\begin{equation*}
  \code{Iion\_cm}
   \;=\;
   \frac{I_{\text{ion}}^{\text{(memb)}}}{C_{m}}
   \;\equiv\;
   \frac{1}{\chi C_{m}}\,
   \bigl(\chi\, I_{\text{ion}}^{\text{(memb)}}\bigr),
\end{equation*}
expressed in V/s, with the electrophysiological convention
\emph{outward (cation efflux) current positive}.
Equation~\eqref{eq:semi-discrete} therefore adds
$-\,\code{Iion\_cm}$ to the PDE source; the external stimulus
\code{externalStimulusCurrent} is stored in A/m$^{3}$ and divided by
$\chi C_{m}$ inside the solver (see \code{createFields.H} and the
\code{rhsVm} assembly in the main solver loop). Throughout the rest
of this report the symbol $\Iion$ is used as a shorthand for
\code{Iion\_cm}; no further rescaling is applied between the cell
model and the explicit update.

\subsection{External stimulus}

\begin{equation}
  \Iext(\mathbf{x}, t) \;=\;
  \begin{cases}
    I_{0} & \mathbf{x} \in \Omega_{\text{stim}}^{(b)},
            \;\; t \in [t_{0}^{(b)},\, t_{0}^{(b)} + \tau], \\
    0     & \text{otherwise},
  \end{cases}
\end{equation}
with Niederer-benchmark defaults
$I_{0} = \SI{50000}{\ampere\per\meter\cubed}$ and
$\tau = \SI{2}{\milli\second}$.

% ================================================================== %
\section{Time discretisation: forward Euler with parabolic CFL bound}
\label{sec:time}
% ================================================================== %

The solver advances the semi-discrete
equation~\eqref{eq:semi-discrete} by a single forward-Euler step on
the whole right-hand side:
\begin{equation}
  \mathsf{V}^{n+1}
   \;=\;
   \mathsf{V}^{n}
   \;+\;
   \Delta t\,
   \Bigl[
     \Lop\,\mathsf{V}^{n}
     \;-\;\Iion^{n}
     \;+\;\Iext^{n}/(\chi C_{m})
   \Bigr],
   \label{eq:euler}
\end{equation}
with $\Lop$ the discrete operator of $\mathcal{L}$ (Section
\ref{sec:space}). No implicit matrix is assembled and no linear
system is solved.

\subsection{Stable time step}
\label{sec:cfl}

The diffusion half of~\eqref{eq:semi-discrete} is parabolic; the
corresponding explicit stability bound is
\begin{equation}
  \Delta t_{\text{CFL}}
   \;=\;
   \text{CFL}\;\,
   \frac{\Delta x^{2}}{d\, D_{\text{eff,max}}},
   \qquad
   D_{\text{eff,max}}
   \;=\;
   \frac{\max\!\bigl(D_{xx}, D_{yy}, D_{zz}\bigr)}{\chi C_{m}},
   \label{eq:cfl}
\end{equation}
where $d$ is the active geometric dimension (\code{mesh.nGeometricD()}
in 1/2/3D), and the characteristic cell size is
\begin{equation}
  \Delta x \;=\;
   \Bigl(
     \frac{
       \text{bbox active volume}
     }{N_{c}}
   \Bigr)^{1/d}
   \quad\text{(\code{characteristicDx})}.
   \label{eq:dx}
\end{equation}
For a uniform Cartesian mesh~\eqref{eq:dx} returns the standard cell
edge. The bound~\eqref{eq:cfl} uses the maximum of the diagonal
conductivity components — sufficient for diagonal-dominated
conductivity tensors of the Niederer-style benchmarks; for strongly
anisotropic or non-diagonal $\Dten$ the bound is conservative.

At solver start the effective step is set to
\begin{equation}
  \Delta t \;=\; \min\!\bigl(\Delta t_{\text{req}},\,\Delta t_{\text{CFL}}\bigr),
\end{equation}
where $\Delta t_{\text{req}}$ is the value of
\code{controlDict.deltaT}; the value is then written back into the
running time database with \code{runTime.setDeltaT(dt)}. The driver
script \code{run\_convergence.py} records both quantities in the
results tables (column \code{dt\_requested} versus
\code{dt\_expectedSolver}); see Section~\ref{sec:results}.

\paragraph{Implication.}
For the configurations reported in Section~\ref{sec:results} the
requested $\Delta t = 10^{-2}$~s is many orders of magnitude above
$\Delta x^{2}/(d D_{\text{eff,max}})$; the solver therefore always
runs at $\Delta t = \Delta t_{\text{CFL}}$ and the
\emph{requested} value only sets an upper bound on the step. The
column \code{dt\_expectedSolver} in the result tables is the actual
step. Tightening or loosening \code{controlDict.deltaT} below the
CFL value is the only way to override the bound, at the obvious cost
of stability.

% ================================================================== %
\section{Spatial discretisation}
\label{sec:space}
% ================================================================== %

\subsection{Standard orthogonal FV Laplacian}
\label{sec:stdFV}

When \code{useHighOrder\_Vm = false} (or in 1D), the diffusion is
evaluated by OpenFOAM's built-in operator
\begin{equation}
  \Lop \mathsf{V}
   \;\approx\;
   \frac{1}{\chi C_{m}}\;
   \text{\code{fvc::laplacian(conductivity, Vm)}},
\end{equation}
i.e.\ the classical orthogonal 5/7-point stencil with linear face
interpolation and the corrected snGrad. \code{zeroGradient} and
\code{empty} patches contribute zero flux; \code{fixedValue} (or the
custom \code{fixedVoltage}) patches contribute the one-sided owner
gradient via the standard FV path.

\subsection{High-order LRE diffusion}
\label{sec:hoVm}

When \code{useHighOrder\_Vm = true} and $d > 1$, the Laplacian is
assembled by an explicit face-quadrature integration
(\code{computeHighOrderLaplacian} in the main solver):
\begin{equation}
  \Phi_{f}
   \;=\;
   \sum_{q \in \mathcal{Q}_{f}}
       w_{q}\, \lvert \mathbf{S}_{f} \rvert\,
       \mathbf{n}_{f} \cdot \Dten_{\text{owner}(f)} \cdot
       \grad_{\!h}\Vm(\mathbf{x}_{fq}),
\end{equation}
with face quadrature points $\mathbf{x}_{fq}$ and weights $w_{q}$
provided by the LRE library~\cite{cardiff2018s4f}. The cell
contribution to the Laplacian is then
\begin{equation}
  (\Lop \mathsf{V})_{i}
   \;=\;
   \frac{1}{\chi C_{m}}\,\frac{1}{|K_{i}|}
   \sum_{f \in \partial K_{i}} s_{if}\,\Phi_{f},
\end{equation}
with $s_{if} = \pm 1$ for owner/neighbour orientation. The cell-loop
divergence is taken with \code{fvc::div(fluxVm\_HO)} after the face
fluxes are written.

\paragraph{Boundary faces in the high-order branch.}
For boundary faces, the LRE face quadrature still evaluates a
reconstructed gradient, but the flux is enforced as follows:
\code{zeroGradient} and \code{empty} patches set
$\Phi_{f} \equiv 0$ (true Neumann zero-flux, even though the LRE
stencil could return a non-zero gradient otherwise);
\code{fixedValue} and \code{fixedVoltage} patches use the LRE-
reconstructed gradient on the owner side. The high-order LRE stencil
is therefore strictly internal in the Neumann-only case used in the
2D/3D benchmarks reported here, and the formal spatial order is
expected to drop to that of the standard branch in the first layer
of cells adjacent to a non-empty patch.

\subsection{Ionic source quadrature}
\label{sec:hoIion}

If \code{useHighOrder\_Iion = false}, the cell-model is integrated
at cell centres:
\begin{equation}
  \overline{\Iion}_{i}
   \;=\; \Iion(\Vmsub{i}, \mathbf{s}_{i}).
\end{equation}

If \code{useHighOrder\_Iion = true} the TNNP states are evolved at
the cell quadrature points $\mathbf{x}_{iq}$ provided by the LRE
library: $\Vm$ is reconstructed at each point by a Taylor expansion
(\code{reconstructVmAtIionIntegrationPoints})
\begin{equation}
  \Vmsub{iq}
   \;=\;
   \Vmsub{i}
   \;+\;
   \grad\Vmsub{i}\cdot(\mathbf{x}_{iq} - \mathbf{x}_{i})
   \;+\;
   \tfrac{1}{2}\,(\mathbf{x}_{iq} - \mathbf{x}_{i})^{\!\top}\!
   \mathbf{H}_{i}\,(\mathbf{x}_{iq} - \mathbf{x}_{i})
   \;+\;
   \tfrac{1}{6}\,\mathbf{T}^{(3)}_{i}\!:\!(\mathbf{x}_{iq} - \mathbf{x}_{i})^{\otimes 3},
\end{equation}
where the Hessian and third-order term are included only if the LRE
order on $\Vm$ is $\ge 2$ or $\ge 3$ respectively. When
\code{useHighOrder\_Vm = false}, only the linear term is used and
the gradient comes from \code{fvc::grad}. The ionic ODE is then
advanced at each quadrature point against this frozen $\Vmsub{iq}$,
and the cell-averaged source is the quadrature
\begin{equation}
  \overline{\Iion}_{i}
   \;=\;
   \frac{\sum_{q \in \mathcal{Q}_{i}} w_{q}\,\Iion(\Vmsub{iq},\mathbf{s}_{iq})}
        {\sum_{q \in \mathcal{Q}_{i}} w_{q}}
   \qquad
   \bigl(\text{\code{averageIionIntegrationPointsToCells}}\bigr).
   \label{eq:iion-quad}
\end{equation}

% ================================================================== %
\section{Per-step operator splitting}
\label{sec:split}
% ================================================================== %

The solver uses a staggered operator-split per outer step. For
$t^{n}\rightarrow t^{n+1}=t^{n}+\Delta t$:

\begin{algorithm}[H]
\caption{Per-step flow of \texttt{highOrderElectroActivationFoamExplicit}.}
\begin{algorithmic}[1]
\State Save $\mathsf{V}^{n}$ from the current field.
\State Apply stimulus to $\Iext$ at $t^{n}$.
\State \textbf{First ionic call (algebraic).} Call
       \code{ionicModel->calculateCurrent} at the current state
       $\mathbf{s}^{n}$ and voltage $\mathsf{V}^{n}$ to obtain
       $\Iion^{n}$. This is an \emph{algebraic} evaluation; the
       state vector is \emph{not} advanced here.
\State Build $\Lop\mathsf{V}^{n}$ (standard FV or HO LRE,
       Section~\ref{sec:space}).
\State Update voltage by forward Euler:
       \begin{equation*}
         \mathsf{V}^{n+1}
          \;=\;
          \mathsf{V}^{n}
          \;+\;
          \Delta t\,\bigl[
            \Lop\mathsf{V}^{n}
            \;-\;\Iion^{n}
            \;+\;\Iext^{n}/(\chi C_{m})
          \bigr].
       \end{equation*}
\State \textbf{Second ionic call (ODE).} Call
       \code{ionicModel->solveODE} from $t^{n}$ to $t^{n+1}$ with
       voltage $\mathsf{V}^{n+1}$ \emph{frozen}, advancing
       $\mathbf{s}^{n}\to\mathbf{s}^{n+1}$. If
       \code{useHighOrder\_Iion}, the ODE is advanced at the cell
       quadrature points with reconstructed $\Vmsub{iq}^{n+1}$, and
       the cell-averaged $\Iion^{n+1}$ is written via
       Eq.~\eqref{eq:iion-quad}.
\State Update activation-time field
       (Section~\ref{sec:activation}).
\State If \code{stopAfterPointActivation}: check the stop cell.
\State Write fields if \code{runTime.outputTime}.
\end{algorithmic}
\end{algorithm}

\paragraph{Order of accuracy in time.}
The split is first-order accurate: the voltage update~\eqref{eq:euler}
is plain forward Euler, and the cell-model is integrated against the
end-of-step voltage rather than a time-centred value. Restoring a
second-order overall scheme would require either a Strang split with
half-steps on the ionic ODE, or replacing the forward Euler by a
Heun-/RK2-type evaluation; neither is implemented.

\paragraph{Why two distinct ionic calls.}
The first call (\code{calculateCurrent}) evaluates the algebraic
$\Iion$ at the current state; this is the source seen by the voltage
update~\eqref{eq:euler}. The second call (\code{solveODE}) advances
the gating and concentration ODEs to $t^{n+1}$ and \emph{also} returns
$\Iion^{n+1}$, which is stored in the field \code{ionicCurrent} for
output and used by the next step. The two calls share no internal
state — the first one does not pre-advance any gate — so the order is
``algebraic source first, then ODE update with the new voltage''.

\section{Ionic ODE integration}
\label{sec:ode}
% ================================================================== %

For each cell or cell QP, the TNNP ODE~\eqref{eq:tnnp} is advanced
by the OpenFOAM adaptive integrator selected via the \code{solver}
key in \code{timeIntegrationProperties}. Default \code{RKF45}~\cite{hairer2}.
The keys read from that file are:

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Key & Meaning & Default \\
\midrule
\code{solver} & ODE integrator & \code{RKF45} \\
\code{absTol} & Absolute tolerance per state & $10^{-9}$ \\
\code{relTol} & Relative tolerance per state & $10^{-6}$ \\
\code{maxSteps} & Cap on sub-steps per outer call & $10^{4}$ \\
\code{initialODEStep} & Initial sub-step (ms) & $10^{-5}$ \\
\code{CFL} & Diffusion CFL coefficient
            (Eq.~\eqref{eq:cfl}) & $0.1$ \\
\bottomrule
\end{tabular}
\end{center}

The outer CFL coefficient sits in the same dictionary purely for
convenience; functionally it is a time-stepping parameter
(Section~\ref{sec:cfl}) and not an ODE-integrator parameter.

% ================================================================== %
\section{TNNP numerical protection}
\label{sec:tnnpProtect}
% ================================================================== %

The TNNP 2004 cell model contains several mathematical features that
become numerical traps when the state variables are driven outside
their physiological window by the outer solver: gate variables that
overshoot $[0,1]$ make the rate-balance equations stiff or
unphysical; intracellular concentrations that touch zero force
$\log(\cdot)$ / $\sqrt{\cdot}$ / $\mathrm{pow}(\cdot)$ to overflow;
very large $|V|$ saturates the $\exp(\cdot)$ tails of the
$\alpha/\beta$ rate constants; and the $L$-type calcium current
$i_{CaL}$ contains a removable singularity at $V=0$ (a
$V/(\exp(V/RT)-1)$ factor, see Eq.~\eqref{eq:app:ical} in
Appendix~\ref{app:tnnp}). The TNNP wrapper provides a guard,
\code{protectState(state, t, qp, phase)}, whose job is to keep the
state vector inside that physiological window at the boundaries of
every external call into the cell-model kernel. The guard is
intentionally minimal: it does \emph{not} fix the cell model, it
only prevents one bad time-step from cascading into NaN/Inf and
corrupting the rest of the simulation.

\subsection{Master switch and configuration}
\label{sec:tnnp:knobs}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Key & Type & Default \\
\midrule
\code{TNNPNumericalProtection} & Switch & \code{false} \\
\code{TNNPClampGates} & Switch & \code{true} \\
\code{TNNPClampVmForRates} & Switch & \code{true} \\
\code{TNNPLogNumericalProtection} & Switch & \code{true} \\
\code{TNNPConcentrationFloor} & scalar (mM) & $10^{-12}$ \\
\code{TNNPVMinForRates} & scalar (mV) & $-200$ \\
\code{TNNPVMaxForRates} & scalar (mV) & $+100$ \\
\code{TNNPVoltageZeroEps} & scalar (mV) & $10^{-8}$ \\
\code{TNNPMaxProtectionLogEntries} & label & $10^{6}$ \\
\bottomrule
\end{tabular}
\end{center}

The master switch is \code{TNNPNumericalProtection}. When it is
\code{false}, \code{protectState} returns immediately and \emph{none}
of the sub-switches have any effect; in particular,
\code{TNNPClampGates} and \code{TNNPClampVmForRates} default to
\code{true} but are inert without the master switch. The default
configuration is therefore \emph{all protection off}; production runs
typically need to set \code{TNNPNumericalProtection true} explicitly.

\subsection{Phases at which the guard fires}
\label{sec:tnnp:phases}

The wrapper invokes the guard at four phases, distinguished by a
string tag passed to \code{protectState} (used only for logging /
accounting; see \S\ref{sec:tnnp:log}):

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Phase tag & Call site & Acts on \\
\midrule
\code{calculateCurrent} & entry of \code{calculateCurrent}
   & the state vector passed in for the algebraic
     $\Iion$ evaluation (step~3 of \S\ref{sec:split}) \\
\code{solveODE\_start}  & entry of \code{solveODE}
   & the integrator's working state vector at $t^{n}$
     (step~6 of \S\ref{sec:split}) \\
\code{solveODE\_end}    & exit of \code{solveODE}
   & the integrator's working state vector at $t^{n+1}$ \\
\code{derivatives}      & inside \code{derivatives()}, every
                          RKF45 sub-step
   & a \emph{copy} of the state vector (does not mutate $y$) \\
\bottomrule
\end{tabular}
\end{center}

All four sites are gated by the single \code{TNNPNumericalProtection}
flag. The \code{derivatives} site has one important asymmetry: it
copies the integrator state and protects the copy, then feeds the
copy into the rate kernel \code{TNNPcomputeRates}. The integrator's
working vector $y$ is never modified mid-step, so the protection only
changes the value of the rates RKF45 sees, not the state it
integrates.

\subsection{Corrections and their order}
\label{sec:tnnp:steps}

When fired, \code{protectState} applies the following corrections, in
this order, to the (possibly copied) state vector:

\begin{enumerate}
\item \textbf{Non-finite $V$:} if \code{state[V]} is \code{NaN} or
\code{Inf}, it is reset to $-86.2$~mV (TNNP resting potential, by
construction inside $[\text{vmMin}, \text{vmMax}]$).
\item \textbf{Voltage clamp for rates} (if \code{TNNPClampVmForRates}):
$V$ is clamped to $[\code{TNNPVMinForRates},\code{TNNPVMaxForRates}]$.
After step~1 this step is a no-op; it only fires when $V$ was finite
but outside the rate window.
\item \textbf{Avoid $i_{CaL}$ singularity:} if $|V| <
\code{TNNPVoltageZeroEps}$, $V$ is pushed to $\pm\code{TNNPVoltageZeroEps}$
(sign preserved). This removes the $V/(\exp(V/RT)-1)$ removable
singularity in $i_{CaL}$ at the symbolic level. Steps~1--2 may have
moved $V$ off zero already; this is the safety net for the case
where $V$ is finite, inside the rate window, but right on zero.
\item \textbf{Concentration floor:} each of
$K_{i}, Na_{i}, Ca_{i}, Ca_{SR}$ is replaced by
\code{TNNPConcentrationFloor} ($10^{-12}$ mM) if it is non-finite or
below the floor. This prevents $\log(\cdot)$ and rational-exponent
operations on near-zero concentrations from overflowing.
\item \textbf{Gate clamp} (if \code{TNNPClampGates}): the twelve
gating variables $Xr_{1}, Xr_{2}, Xs, m, h, j, d, f, fCa, s, r, g$
are clamped to $[0, 1]$ (with non-finite values pulled to $0$). This
is the most frequent correction in practice because adaptive ODE
sub-stepping can overshoot the bounds by $\mathcal{O}(\code{relTol})$
between $t^{n}$ and $t^{n+1}$.
\end{enumerate}

Each step is idempotent and operates on the already-corrected state,
so a single \code{protectState} call is enough — there is no inner
fixed-point iteration. The ionic currents are computed from the
corrected state after step~5.

\subsection{Interaction with the RKF45 step controller}
\label{sec:tnnp:rkf45}

The \code{derivatives} call site introduces \emph{non-smoothness} in
the right-hand side seen by RKF45: clamping is $C^{0}$ at best
(piecewise linear in the offending component), and the gate clamp is
a saturation that flattens the Jacobian once the gate touches its
bound. The RKF45 embedded error estimator implicitly assumes
$C^{q+1}$ smoothness of the RHS over the sub-step, so firing the
guard inside \code{derivatives()} can inflate the local error
estimate around clamp boundaries and trigger spurious sub-stepping.
Because the integrator's working state $y$ is \emph{not} mutated
(\S\ref{sec:tnnp:phases}), the practical effect is bounded: only the
rates are perturbed, the state recovers as soon as the active gate
leaves the clamp. Even so, a high count of \code{phase=derivatives}
corrections in the summary file is a diagnostic signal that the cell
ODE is being driven outside its valid regime — usually by a too-large
$\Delta t$ (note: the outer $\Delta t$ here is already bounded by CFL,
so this is more likely a transient in $\Vm$ that the explicit
diffusion solver could not follow). The recommended response is to
investigate the upstream cause; the \code{derivatives}-phase
protection is a safety net, not a correction mechanism.

\subsection{What the guard does not protect}
\label{sec:tnnp:notProtected}

The five corrections above address NaN/Inf propagation, gate
excursion outside $[0,1]$, the $V=0$ singularity of $i_{CaL}$, and
concentration underflow. They do \emph{not} address the following
remaining failure modes intrinsic to the TNNP cell model:

\begin{itemize}[leftmargin=*]
\item Removable singularities in the $\alpha/\beta$ rate expressions
of $i_{Na}$ and $i_{Kr}$ (notably the $1/(\exp(\cdot)-1)$ factors at
$V = -47.13$~mV and similar). These can be regularised
\emph{analytically} by L'H\^opital substitution near the singular
point; the present wrapper leaves them to the adaptive integrator.
\item Stiff blow-ups in the calcium-handling subsystem when $Ca_{SR}$
approaches the floor or when $fCa$ saturates close to $1$. Clamping
$Ca_{SR}$ avoids overflow but does not regularise the
$Ca_{i}/Ca_{SR}$ ratio that drives the SR-release flux.
\item Exponential overflow of the $\alpha/\beta$ tails for $|V| \gg
\text{vmMax}$. Clamping $V$ to the rate window is a workaround; the
true fix is to ensure $V$ never leaves the physiological window in
the first place (i.e.\ pick a CFL/step combination that keeps the
upstroke under control).
\end{itemize}

In short: the guard turns a numerical disaster (NaN cascade) into a
recoverable mis-evaluation. It is not a substitute for an
adequately-resolved outer solver.

\subsection{Logging and accounting}
\label{sec:tnnp:log}

Each invocation of \code{correctStateComponent} records a triplet
\code{(phase, variable, reason)} together with the offending value,
its corrected value, and the simulation time. If
\code{TNNPLogNumericalProtection} is true, the entries are
accumulated in memory (capped by \code{TNNPMaxProtectionLogEntries})
and emitted on solver exit to the file

\smallskip
\code{TNNP\_numericalProtection\_summary.dat}
\smallskip

\noindent in the case directory. The summary groups corrections by
the \code{(phase, variable, reason)} key, with: count, first
occurrence time, the first old/corrected value pair, and the
min/max/$|\Delta|_{\max}$ over all corrections in the group. Typical
diagnostic use:

\begin{itemize}[leftmargin=*]
\item Many \code{Vm\_below\_minimum} at one specific cell ID
$\Rightarrow$ suspect a Dirichlet BC pulling $V$ outside the rate
window.
\item Many \code{concentration\_below\_floor} after a long simulation
$\Rightarrow$ ODE tolerances too loose; tighten \code{relTol}.
\item Many \code{phase=derivatives} entries during the upstroke
$\Rightarrow$ the inner ODE integrator is straining against the
explicit voltage update; consider tightening the CFL coefficient or
shrinking the requested $\Delta t$ below
$\Delta t_{\text{CFL}}$ to recover head-room.
\item Any \code{nonfinite\_V} or \code{nonfinite\_gate}
$\Rightarrow$ a real divergence already happened upstream; the guard
contained the damage but the result is suspect.
\end{itemize}

\label{sec:tnnp:settings}

The defaults are intentionally conservative (master switch
\code{false}) to keep the wrapper transparent in test configurations.
For production and convergence studies the recommended setting is:

\section{Activation tracking and benchmarks}
\label{sec:activation}
% ================================================================== %

\paragraph{Activation time per cell.} First-crossing of $V_{\text{thr}}$
in a step is recorded by linear interpolation:
\begin{equation}
  w \;=\; \mathrm{clip}\!\Bigl(
      \frac{V_{\text{thr}} - V^{n}}{V^{n+1} - V^{n}},
      \; 0, 1
    \Bigr),
   \qquad
   t_{\text{act}} \;=\; t^{n} + w\, \Delta t.
\end{equation}
The per-cell flag is cleared so re-crossings are ignored.
Default threshold $V_{\text{thr}} = 0$~V (= 0~mV; the Niederer
convention).

\paragraph{Early termination on P8.} When
\code{stopAfterPointActivation = true}, the cell nearest
\code{stopActivationPoint} (P8 in the Niederer benchmark) is
monitored, and the effective end time is set to
$t_{\text{act}} + $~\code{stopDelayAfterActivation}.

\paragraph{Diagonal sampling.} \code{sampleIDW} performs
inverse-distance-weighted interpolation at \code{nDiagonalSamples}
points between \code{diagonalStart} and \code{diagonalEnd} with
$k = $~\code{sampleNeighbours}, $p = $~\code{samplePower}.

\section{Niederer-style benchmark results}
\label{sec:results}
% ================================================================== %

This section collects measured P8 activation times, wall-clock cost
and peak memory.

\paragraph{Scope and disclaimer.}
The published Niederer et al.~2012~\cite{niederer2012} benchmark is
three-dimensional (20$\times$7$\times$3~mm anisotropic slab with
fibre rotation). The 2D case used here is a planar reduction
(20$\times$8~mm, single cell in $z$, planar fibres) defined for
code-development testing only and not reflected, to our knowledge,
in any published reference. The 3D case follows the Niederer slab
geometry (20$\times$7$\times$3~mm) but the sweep is sparser (fewer
LRE pairs and mesh sizes) than the 2D one. The numbers below should
therefore be read as an \emph{internal} self-consistency study
($h$- and $\Delta t$-refinement, plus the effect of the LRE
$p$-levels) and not as a comparison with any external reference
value.

Rows of the tables below are LRE p-levels for $\Vm$ $\times$ $\Iion$
(\texttt{NO} = high-order disabled); columns are mesh sizes
$\Delta x$ in millimetres. Computational cost is the elapsed
wall-clock time of the solver process reported by
\texttt{/usr/bin/time -v}; peak memory is the maximum resident set
size of the same process. \textsc{NC} = no run data found for that
combination.

\paragraph{Reading $\Delta t_{\text{req}}$ versus $\Delta t_{\text{CFL}}$.}
The forward-Euler scheme is bounded by the parabolic CFL condition
(Section~\ref{sec:cfl}). The 2D sweep was driven with a single
$\Delta t_{\text{req}} = 10^{-2}$~s; this value is much larger than
the CFL bound at all mesh sizes used, so the solver always integrates
with $\Delta t = \Delta t_{\text{CFL}}$ — the values are listed in
the header table of \S\ref{sec:results:2D}. In the 3D sweep
$\Delta t_{\text{req}}$ is set close to the CFL bound, so for some
configurations the CFL clamp \emph{does not} bind and the solver
runs at the requested step; for others it does. Both columns are
reported.

\subsubsection{Discussion of the 2D results}

\paragraph{$h$-refinement at fixed configuration.}
For the low-order pair (\code{NO}, \code{NO}), the P8 activation
time decreases monotonically with mesh refinement
($160.63\rightarrow 68.12\rightarrow 50.35\rightarrow 48.15\rightarrow
45.62$~ms from $\Delta x = 0.5$ to $0.05$~mm). At each $\Delta x$ the
effective step also decreases as $\Delta x^{2}$
($1.3\!\cdot\!10^{-4}\rightarrow 1.3\!\cdot\!10^{-6}$~s); the
forward-Euler scheme is therefore well below its stability bound
\emph{by construction} on every line of the table. The asymptotic
value at the finest mesh is $\approx 45.6$~ms.

\paragraph{High-order on the source.}
Activating the LRE quadrature on $\Iion$ — without changing the
diffusion (\code{NO}, \code{p2}) — drops the coarsest-mesh value
from $160.63$~ms to $34.71$~ms and brings the finer-mesh values to
$\approx 45$~ms already at $\Delta x = 0.125$. The high-order ionic
quadrature is therefore the most cost-effective single switch for
accuracy on coarse meshes; see Eq.~\eqref{eq:iion-quad} for the
mechanism.

\paragraph{High-order on the diffusion alone.}
The combinations (\code{p1}, \code{NO}), (\code{p2}, \code{NO}),
(\code{p3}, \code{NO}) — high-order LRE diffusion with cell-centre
ionic source — are noticeably less stable on the coarsest mesh:
\code{p1/NO} at $\Delta x=0.5$~mm returns $14.25$~ms (clearly a
spurious very-early crossing), and \code{p2/NO} returns $0.00$~ms
(the front never crossed within the simulated window, or crossed
spuriously at $t=0$). These values are diagnostic, not physical:
the high-order Laplacian operating on a piecewise-constant ionic
source produces an over-shooting voltage on coarsely-resolved fronts,
which the activation-time tracker then latches at the first numerical
crossing. The pathology disappears once the ionic source is also
volume-integrated at LRE quadrature points (any \code{p$k$}/\code{p$k$}
pair gives a well-behaved value on the same mesh).

\paragraph{Full HO at the finer meshes.}
The diagonal pairs (\code{p2}, \code{p2}) and (\code{p3}, \code{p3})
plateau quickly: at $\Delta x = 0.125$~mm and $\Delta x = 0.1$~mm
the values are $48.5$/$48.1$~ms and $45.5$/$45.7$~ms respectively;
at $\Delta x = 0.05$~mm the (\code{p2}, \code{p2}) value is
$46.6$~ms. The off-diagonal pairs
(\code{NO}, \code{p2}), (\code{NO}, \code{p3}),
(\code{p1}, \code{p2}), (\code{p1}, \code{p3}),
(\code{p3}, \code{p2}), (\code{p3}, \code{p3}) all collapse to a
mutually-consistent value in $[44.5, 46.0]$~ms at $\Delta x \le 0.1$~mm.
This $[44.5, 46.0]$~ms band is the internal-consistency floor of
the 2D dataset; we do not claim it represents the ``true'' value
for this 2D problem.

\paragraph{Cost.}
Wall-clock cost is dominated by the
$N_{c}\times(\text{ionic-ODE work per cell})$ product and by the
inverse $\Delta t_{\text{CFL}}$. Refining the mesh from
$\Delta x = 0.5$ to $0.05$~mm increases the cell count by $10^{2}$ and
the step count by $\sim 10^{2}$, so the total cost grows by
$\sim 10^{4}$ — consistent with the (\code{NO}, \code{NO}) entries
of the wall-clock table (1~min vs.\ 168~min).
Peak RSS grows roughly linearly with $N_{c}$, with HO levels adding a
constant-factor overhead for the LRE stencils and the per-quadrature-
point state vectors.

\subsubsection{Discussion of the 3D results}

\paragraph{$\Delta t_{\text{req}}$ versus $\Delta t_{\text{CFL}}$.}
In the 3D sweep some rows have
$\Delta t_{\text{req}} = \Delta t_{\text{CFL}}$ (the requested
step happened to be at or below the CFL bound, so the solver
integrated with that value), while others have
$\Delta t_{\text{req}} > \Delta t_{\text{CFL}}$ (the CFL clamp
binds, the solver integrates with the smaller CFL value). Both
columns are reported because they are dictated by different
constraints: $\Delta t_{\text{req}}$ comes from the user's
\code{controlDict}, $\Delta t_{\text{CFL}}$ from the
mesh-and-conductivity product.

\paragraph{Coarse-mesh sensitivity.}
At the coarsest mesh ($\Delta x = 0.5$~mm,
40$\times$6$\times$14 cells), the low-order (\code{NO},
\code{NO}) configuration gives $t_{\text{act}}^{P8} \approx 141$~ms
— consistent across both $\Delta t = 10^{-5}$ and $5\!\cdot\!10^{-6}$~s
(140.96 vs.\ 140.91 ms): the temporal error is small but the
spatial error is dominant. Adding even \code{p1} to the ionic
quadrature collapses the value to $\approx 36.7$~ms at the same
mesh, with the same temporal stability. The (\code{p3}, \code{p3})
pair reaches $\approx 33$~ms on the same coarse mesh.

\paragraph{One mesh refinement.}
For the rows that exist at $\Delta x = 0.25$~mm (with
$\Delta t_{\text{CFL}} \approx 2.2\!\cdot\!10^{-5}$~s), the
high-order pairs cluster in $[40.3, 46.4]$~ms across the four LRE
combinations available. The (\code{p3}, \code{p3}) pair value
$40.27$~ms is the lowest of this group.

\paragraph{Where the 2D and 3D values disagree.}
The 2D and 3D problems are different (different geometry, different
ratio of diffusion to reaction), so the activation times are not
comparable directly. The qualitative pattern is the same in both
cases: high-order on the source (\code{p}$k$, second slot) is the
single biggest accuracy lever; mesh-only refinement with low-order
diffusion converges from above (the front arrives later than the
asymptotic value); turning on high-order diffusion in isolation
(\code{p}$k$, \code{NO}) is unstable on coarse meshes.

\subsection{Conclusions on the 2D/3D study}
\label{sec:results:conclusions}

\begin{enumerate}[leftmargin=*]
\item The forward-Euler scheme is always run at $\Delta t =
\Delta t_{\text{CFL}}$ when $\Delta t_{\text{req}} \gg
\Delta t_{\text{CFL}}$ (the 2D sweep) and at $\Delta t =
\Delta t_{\text{req}}$ otherwise (some of the 3D rows). The two
columns in the result tables are not redundant; they reflect a
\code{min()} clamp in the solver and the actually-used
step is the smaller of the two.
\item The LRE quadrature on the ionic source
(\code{useHighOrder\_Iion}) is the most cost-effective accuracy
switch on both 2D and 3D meshes; on coarse meshes it brings the
P8 activation time from $\sim 160$~ms (2D) / $\sim 140$~ms (3D)
to physical values $\sim 40$~ms.
\item The high-order LRE diffusion in isolation
(\code{useHighOrder\_Vm} with cell-centre ionic source) is
\emph{not} robust on coarse meshes; the high-order reconstruction
of the Laplacian on top of a piecewise-constant source leads to
non-physical early crossings. The recommended use is to enable
both switches together, or to keep the low-order branch on the
diffusion when the source is also low-order.
\item The 2D internal-consistency floor (best mutually-consistent
band across high-order pairs at the finer meshes) sits at
$[44.5, 46.0]$~ms; the 3D high-order pairs at $\Delta x = 0.25$~mm
sit in $[40.3, 46.4]$~ms. We do not claim either band represents
the ``true'' value for the respective problem; published 3D
Niederer values are reported with a wide cross-code spread.
\item For matched accuracy, the explicit forward-Euler scheme is
much more expensive than the implicit Crank-Nicolson scheme of the
companion solver, because $\Delta t_{\text{CFL}} = \mathcal{O}
(\Delta x^{2})$ binds and the per-step cost is dominated by the
cell-model ODE work (which is independent of the time
discretisation). The explicit solver is competitive on coarse
meshes and as a verification baseline for the implicit solver,
not as a production tool on fine meshes.
\end{enumerate}

\section{TNNP rate expressions (verbatim)}
\label{app:tnnp}
% ================================================================== %

For completeness, the rate expressions used by the TNNP 2004 cell
model wrapper are reproduced below. The membrane voltage entering
these expressions is in millivolts ($V = 10^{3}\,\Vm$ when the
PDE-side voltage is in volts); concentrations are in mM; time
constants are in ms. References are to \cite{ttp2004}.

\paragraph{Reversal potentials.}
\begin{align}
  E_{Na} &= \frac{RT}{F}\ln\left(\frac{Na_o}{Na_i}\right),\\
  E_K   &= \frac{RT}{F}\ln\left(\frac{K_o}{K_i}\right),\\
  E_{Ks} &=
  \frac{RT}{F}
  \ln\left(\frac{K_o + P_{kna}Na_o}{K_i + P_{kna}Na_i}\right),\\
  E_{Ca} &= \frac{1}{2}\frac{RT}{F}\ln\left(\frac{Ca_o}{Ca_i}\right).
\end{align}

\paragraph{Total ionic current.}
\begin{equation}
  I_{\text{ion}} =
  i_{K1}+i_{to}+i_{Kr}+i_{Ks}+i_{CaL}+i_{NaK}
  +i_{Na}+i_{bNa}+i_{NaCa}+i_{bCa}+i_{pK}+i_{pCa}.
\end{equation}

\paragraph{Individual currents.}
\begin{align}
  i_{Na} &= g_{Na}m^3hj(V-E_{Na}),\\
  i_{bNa} &= g_{bNa}(V-E_{Na}),\\
  i_{K1} &= g_{K1}x_{K1,\infty}\sqrt{K_o/5.4}(V-E_K),\\
  i_{to} &= g_{to}rs(V-E_K),\\
  i_{Kr} &= g_{Kr}\sqrt{K_o/5.4}X_{r1}X_{r2}(V-E_K),\\
  i_{Ks} &= g_{Ks}X_s^2(V-E_{Ks}),\\
  i_{bCa} &= g_{bCa}(V-E_{Ca}),\\
  i_{pK} &= \frac{g_{pK}(V-E_K)}{1+\exp((25-V)/5.98)},\\
  i_{pCa} &= \frac{g_{pCa}Ca_i}{Ca_i+K_{pCa}}.
\end{align}
\begin{equation}
  i_{CaL}
  =
  \frac{g_{CaL}\,d\,f\,f_{Ca}\,\dfrac{4VF^{2}}{RT}
       \left(Ca_i\exp(2VF/RT)-0.341\,Ca_o\right)}
       {\exp(2VF/RT)-1}.
  \label{eq:app:ical}
\end{equation}
\begin{align}
  i_{NaK} &=
  \frac{P_{NaK}\dfrac{K_o}{K_o+K_{mk}}\dfrac{Na_i}{Na_i+K_{mNa}}}
       {1+0.1245\exp(-0.1VF/RT)+0.0353\exp(-VF/RT)},\\
  i_{NaCa} &=
  \frac{K_{NaCa}\left[\exp(\gamma VF/RT)Na_i^{3}Ca_o
       -\exp((\gamma-1)VF/RT)Na_o^{3}Ca_i\alpha\right]}
       {(K_{mNai}^{3}+Na_o^{3})(K_{mCa}+Ca_o)
        \left[1+K_{sat}\exp((\gamma-1)VF/RT)\right]}.
\end{align}

\paragraph{Gate kinetics (selected; full set generated by the
TNNP code generator).}
Most gates use the standard $x_\infty/\tau_x$ form
$\dot x = (x_\infty - x)/\tau_x$. The implemented steady states and
time constants for $m, h, j, d, f, X_{r1}, X_{r2}, X_s, r$ are the
ones of \cite{ttp2004} and are not reproduced here in full. The
$s$ gate depends on tissue type:
\begin{equation}
  s_\infty =
  \begin{cases}
  \left[1+\exp((V+20)/5)\right]^{-1},
  & \text{endocardial},\\
  1.1\left[1+\exp((V+28)/6)\right]^{-1},
  & \text{M cell or epicardial},
  \end{cases}
\end{equation}
with $\tau_s$ also tissue-dependent. The calcium-dependent gates
follow
\begin{align}
  \alpha_{fCa} &= [1+(Ca_i/0.000325)^{8}]^{-1},\\
  \beta_{fCa} &= 0.1[1+\exp((Ca_i-0.0005)/0.0001)]^{-1},\\
  \beta_d &= 0.2[1+\exp((Ca_i-0.00075)/0.0008)]^{-1},\\
  f_{Ca,\infty} &= (\alpha_{fCa}+\beta_{fCa}+\beta_d+0.23)/1.46,
\end{align}
with the conditionally-frozen update
\begin{equation}
  \frac{df_{Ca}}{dt}
  =
  \begin{cases}
  0, & f_{Ca,\infty}>f_{Ca}\ \text{and}\ V>-60,\\
  (f_{Ca,\infty}-f_{Ca})/\tau_{fCa}, & \text{otherwise},
  \end{cases}
\end{equation}
and analogously for the RyR $g$ gate.

\paragraph{Concentrations and SR fluxes.}
\begin{align}
  \frac{dNa_i}{dt} &=
   -\frac{(i_{Na}+i_{bNa}+3i_{NaK}+3i_{NaCa})C_m}{V_cF},\\
  \frac{dK_i}{dt} &=
   -\frac{[(i_{K1}+i_{to}+i_{Kr}+i_{Ks}+i_{pK}+I_{stim})
   -2i_{NaK}]C_m}{V_cF},\\
  i_{rel} &=
   \left[\frac{a_{rel}Ca_{SR}^{2}}{b_{rel}^{2}+Ca_{SR}^{2}}+c_{rel}\right]dg,\\
  i_{up} &= \frac{V_{max,up}}{1+K_{up}^{2}/Ca_i^{2}},\\
  i_{leak} &= V_{leak}(Ca_{SR}-Ca_i),\\
  J_{Ca_i} &=
   -\frac{(i_{CaL}+i_{bCa}+i_{pCa}-2i_{NaCa})C_m}{2V_cF}
   + i_{leak} - i_{up} + i_{rel},\\
  \frac{dCa_i}{dt}    &= J_{Ca_i}\,f_{J Ca_i},\\
  \frac{dCa_{SR}}{dt} &= \frac{V_c}{V_{SR}}(i_{up}-i_{rel}-i_{leak})\,f_{J Ca_{SR}},
\end{align}
with buffering factors $f_{J Ca_i}, f_{J Ca_{SR}}$ of the form
$[1+\text{Buf}\cdot K/(\text{Ca}+K)^{2}]^{-1}$.
