\section{Governing Equation}
The solver integrates the transient diffusion equation:
\begin{equation}
\rho C_p\,\frac{\partial T}{\partial t} = \nabla\cdot(\alpha\nabla T) + Q,
\end{equation}
with constant material coefficients ($\alpha=1$, $\rho=1$, $C_p=1$ in the current case setup).
In code, the spatial residual is defined as $R = \nabla\cdot(\alpha\nabla T)+Q$, and the explicit update is:
\begin{equation}
T_i^{n+1}=T_i^n + \frac{\Delta t}{\rho C_p}\,R_i^n.
\end{equation}

\section{Numerical Method}
\subsection{Space: high-order LRE}
When \texttt{useHighOrder\_T=true}, diffusive face fluxes are integrated by face quadrature using LRE-reconstructed gradients (\texttt{gradScalarFaceQuad}, \texttt{faceQuadWeight}), and assembled with \texttt{fvc::div(fluxT\_HO)}.

\subsection{Time: explicit Euler}
No implicit temperature linear solve is performed at each time step; integration is fully explicit at the cell level.
In semi-discrete form:
\begin{equation}
\mathbf{M}\,\dot{\mathbf{T}} = \mathbf{R}(\mathbf{T},t), \qquad \mathbf{M}_{ii}=\rho C_p V_i,
\end{equation}
i.e. a diagonal (lumped) mass matrix naturally arising from cell-centered FV. A consistent non-diagonal mass matrix is not used here.

\subsection{Boundary stencils}
In \texttt{createFields.H}, \texttt{includePatchInStencils} enables boundary contribution for \texttt{fixedValue}, \texttt{fixedVoltage}, and \texttt{zeroGradient} patches.
\section{Errors and Convergence Orders}
\subsection{Cell-centered errors}
With $e_i=T_i-T_{\mathrm{ex},i}$:
\begin{align}
L_1 &= \frac{\sum_i |e_i|V_i}{\sum_i V_i}, \\
L_2 &= \sqrt{\frac{\sum_i e_i^2V_i}{\sum_i V_i}}, \\
L_\infty &= \max_i |e_i|, \\
\texttt{error\_T} &= \sqrt{\sum_i e_i^2}.
\end{align}

\subsection{Gauss-point reconstructed errors}
The solver reconstructs $T_h$ at cell quadrature points (using gradient/Hessian/third derivative according to LRE order) and computes $G_1$, $G_2$, $G_\infty$ analogously.

\subsection{Spatial and temporal rates}
For each fixed $\Delta t$, spatial rates are estimated from fits $e\sim h^p$ (reported in \texttt{spatial\_convergence\_rate\_dt\_...\_total.dat}) and pairwise slopes between consecutive meshes (reported in \texttt{...\_pairwise.dat}).
Temporal rates are estimated from finest-mesh errors ($N=30$) via $e\sim (\Delta t)^{p_t}$.
