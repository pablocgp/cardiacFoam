\section{Governing Equation and MMS}
The solver addresses the stationary diffusion problem in $\Omega=[0,1]^3$:
\begin{equation}
\nabla\cdot\left(\alpha\nabla T\right) + Q = 0,
\end{equation}
with constant diffusivity $\alpha$ (here, $\alpha=1$).

In MMS, an exact temperature $T_{\mathrm{ex}}$ is prescribed and the source is set as:
\begin{equation}
Q = -\nabla\cdot\left(\alpha\nabla T_{\mathrm{ex}}\right).
\end{equation}

In this solver, for all five implemented examples, $Q$ is proportional to $T_{\mathrm{ex}}$ (Table~\ref{tab:mms-defs}).

\section{Numerical Discretization}
\subsection{Spatial Operator}
A high-order diffusive face-flux approximation is used via LRE (Least-Squares Reproducing Kernel) and face quadrature:
\begin{equation}
\phi_f = \int_{A_f} \mathbf{n}_f\cdot(\mathbf{K}\nabla T)\,dA \approx A_f\sum_q w_{f,q}\,\mathbf{n}_f\cdot(\mathbf{K}\nabla T)_q,
\end{equation}
with $\mathbf{K}=\alpha\mathbf{I}$. Then $\nabla\cdot(\mathbf{K}\nabla T)$ is assembled with \texttt{fvc::div(fluxT\_HO)}.

When \texttt{useHighOrder\_T=false}, the solver falls back to \texttt{fvc::laplacian(conductivity, T)}.

\subsection{Stationary Solution Strategy (Pseudo-Time)}
There is no physical time derivative $\partial T/\partial t$. The steady problem is solved by explicit pseudo-time relaxation:
\begin{equation}
T^{k+1} = T^k + \Delta\tau\,R^k, \qquad R^k = \nabla\cdot(\alpha\nabla T^k)+Q.
\end{equation}

The pseudo-time step is:
\begin{equation}
\Delta\tau = \texttt{pseudoCFL}\,\frac{\Delta x^2}{\alpha}.
\end{equation}
Iterations stop when $\lVert R \rVert_{\infty}<\texttt{residualTol}$ (or \texttt{maxPseudoSteps} is reached).

\paragraph{Important note on mass matrices and implicit/explicit time integration.}
This stationary solver has no mass term (no $dT/dt$), therefore there is no consistent/lumped mass matrix and no physical-time implicit integration. The method is explicit only in pseudo-time, as an iterative route to the steady state.

\subsection{Boundary Stencils}
In \texttt{createFields.H}, \texttt{includePatchInStencils} is enabled only for \texttt{fixedValue}/\texttt{fixedVoltage} patches.
\texttt{zeroGradient} patches are excluded from boundary LRE stencils. This can affect near-boundary accuracy and global rates (especially in $L_\infty$-type norms).

\section{Error Definitions and Convergence Orders}
\subsection{Cell-Centered Errors}
With $e_i = T_i - T_{\mathrm{ex},i}$ and cell volume $V_i$:
\begin{align}
L_1 &= \frac{\sum_i |e_i|V_i}{\sum_i V_i}, \\
L_2 &= \sqrt{\frac{\sum_i e_i^2 V_i}{\sum_i V_i}}, \\
L_\infty &= \max_i |e_i|.
\end{align}

The solver also reports \texttt{error\_T} as:
\begin{equation}
\texttt{error\_T}=\sqrt{\sum_i e_i^2},
\end{equation}
which is not volume-normalized. On uniform 3D meshes, its expected slope approximately follows
\begin{equation}
p_{\texttt{error\_T}} \approx p_{L_2}-\tfrac{3}{2}.
\end{equation}

\subsection{Gauss-Point Reconstructed Errors}
At each cell quadrature point $\mathbf{x}_q$, the solver reconstructs:
\begin{equation}
T_h(\mathbf{x}_q)=T_C+\nabla T_C\cdot\mathbf{d}+\frac{1}{2}\mathbf{d}^T\mathbf{H}_C\mathbf{d}+\frac{1}{6}\mathcal{T}_C(\mathbf{d},\mathbf{d},\mathbf{d}),
\end{equation}
with $\mathbf{d}=\mathbf{x}_q-\mathbf{x}_C$. Hessian and third-derivative terms are included when the LRE order supports them.

Then:
\begin{align}
G_1 &= \frac{\sum_{i,q}V_i w_{iq}|T_h(\mathbf{x}_{iq})-T_{\mathrm{ex}}(\mathbf{x}_{iq})|}{\sum_{i,q}V_i w_{iq}}, \\
G_2 &= \sqrt{\frac{\sum_{i,q}V_i w_{iq}(T_h-T_{\mathrm{ex}})^2}{\sum_{i,q}V_i w_{iq}}}, \\
G_\infty &= \max_{i,q}|T_h(\mathbf{x}_{iq})-T_{\mathrm{ex}}(\mathbf{x}_{iq})|.
\end{align}

\subsection{Convergence Rates (\texttt{run\_convergence.py})}
Using $h=1/N$, for two consecutive meshes $(N_1,N_2)$:
\begin{equation}
p_{N_1\to N_2}=\frac{\log(e_1/e_2)}{\log(h_1/h_2)}=\frac{\log(e_1/e_2)}{\log(N_2/N_1)}.
\end{equation}
Additionally, a global fitted slope $p_{\mathrm{fit}}$ is computed by log--log linear regression over all meshes.
